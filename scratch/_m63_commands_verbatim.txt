import numpy as np, pandas as pd, sys
sys.path.insert(0,"Model 5/Model 5 Experiment 1")
import transfer_eval as te
F=list(te.FEATURES); IMG=te.image_features(".")
def pair_test(a,b):
    A=IMG[(IMG.sample_id==a)&(IMG.split=="train")][F].to_numpy(float)
    B=IMG[(IMG.sample_id==b)&(IMG.split=="train")][F].to_numpy(float)
    out=[]
    for j,f in enumerate(F):
        x,y=A[:,j],B[:,j]
        s=np.sqrt(((x.var(ddof=1)/len(x))+(y.var(ddof=1)/len(y))))
        t=(x.mean()-y.mean())/(s+1e-12)
        # overlap of the two per-image distributions: can ANY threshold separate them?
        lo,hi=min(x.mean(),y.mean()),max(x.mean(),y.mean())
        frac=np.mean((x>min(x.mean(),y.mean()))&(x<max(x.mean(),y.mean())))
        out.append((f,t,abs(x.mean()-y.mean())/(np.std(np.r_[x,y])+1e-12)))
    ts=np.array([o[1] for o in out]); ds=np.array([o[2] for o in out])
    return ts,ds,out
pairs=[("H038","G190"),("H037","H368"),("H405","H615"),("F827","H372"),("H616","H183")]
print("Can the two members of each ALIASING PAIR be separated at all, from per-image features?")
print("(a feature exists only if some column shows between-soil difference >> within-soil scatter)")
for a,b in pairs:
    ts,ds,_=pair_test(a,b)
    k=int(np.argmax(ds))
    print("  %-6s vs %-6s : max Cohen-like d = %5.2f (feature %s) | features with |t|>3: %2d/12 | median d %.2f"%(
        a,b,ds.max(),F[k],int((np.abs(ts)>3).sum()),np.median(ds)))
print("""
  Reading guide: d ~ 0 means the two soils' images are statistically the same in every feature --
  no function of these images can tell them apart, so the error is destroyed before the model.
  d > ~1.5 means a real, findable separating feature exists.""")
# aggregate: how separable are aliasing pairs vs well-modelled pairs
def dsum(a,b):
    ts,ds,_=pair_test(a,b); return ds.max()
al=[dsum(*p) for p in pairs]
ok=[dsum("H368","H615"),dsum("H181","H183"),dsum("H372","H181"),dsum("H615","G190")]
print("\n  aliasing-pair max-d:  "+" ".join("%.2f"%v for v in al)+"   (median %.2f)"%np.median(al))
print("  low-error-pair max-d: "+" ".join("%.2f"%v for v in ok)+"   (median %.2f)"%np.median(ok))
